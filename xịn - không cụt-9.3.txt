    // Chuẩn hoá cách bỏ dấu (hoá/hóa, thuý/thúy...) để so khớp chuỗi với kho từ
    function toneSyl(sy) {
        sy = (sy || '').normalize('NFC');
        const m = sy.match(/^(.*?)(o[áàảãạéèẻẽẹ]|u[ýỳỷỹỵ])$/);
        if (!m) return sy;
        if (m[2][0] === 'u' && m[1].endsWith('q')) return sy;
        return m[1] + TONE_FIX[m[2]];
    }
    function toneKey(w) {
        return (w || '').trim().toLowerCase().replace(/\s+/g, ' ').split(' ').map(toneSyl).join(' ');
    }

    function gtFindValidCombos(syllables) {
        if (!byFirst.size) return [];
        const lowerSyllables = syllables.map(s => s.toLowerCase());
        const results = [];
        const seen = new Set();
        for (const t1 of lowerSyllables) {
            const opts = byFirst.get(t1);
            if (!opts) continue;
            for (const full of opts) {
                const t2 = full.split(' ')[1];
                if (lowerSyllables.includes(t2) && !seen.has(full)) {
                    seen.add(full);
                    results.push(full);
                }
            }
        }
        return results;
    }

    function normWord(w) {
        return (w || '').trim().toLowerCase().replace(/\s+/g, ' ');
    }

    function buildIndex(entries) {
        byFirst.clear();
        trapSet.clear();
        byLast.clear();
        keyToWord.clear();
        chainDictWords.clear();
        const validSet = new Set();
        for (const entry of entries) {
            const word = (typeof entry === 'string') ? entry : entry.w;
            const trap = (typeof entry === 'object' && entry) ? !!entry.trap : false;
            if (!word) continue;
            const norm = normWord(word);
            const parts = norm.split(' ');
            if (parts.length !== 2) continue;
            validSet.add(norm);
            if (trap) trapSet.add(norm);
            const first = parts[0];
            if (!byFirst.has(first)) byFirst.set(first, []);
            byFirst.get(first).push(norm);
            const wk = toneKey(norm);
            const wkp = wk.split(' ');
            if (!byLast.has(wkp[1])) byLast.set(wkp[1], []);
            byLast.get(wkp[1]).push({ w: norm, p: wkp[0] });
            if (!keyToWord.has(wk)) keyToWord.set(wk, norm);
            if (chainIdx.has(wk)) chainDictWords.add(norm);
        }
        totalValidWords = validSet.size;
    }

    // ===== Gom Từ: quét 16 ô và ghép từ 2 âm tiết =====
    let gtLastKey = "";

    function gtScan() {
        const panel = document.getElementById('wd-mobile-panel');
        if (!panel) return;

        const allElements = document.querySelectorAll('*');
        let detectedWords = [];

        for (let el of allElements) {
            if (el.closest('#wd-mobile-panel') || el.closest('#wd-mini-panel')) continue;
            if (el.children.length === 0 && el.innerText) {
                let text = el.innerText.trim();
                text = text.replace(/[^a-zA-ZàáảãạăắằẳẵặâấầẩẫậèéẻẽẹêếềểễệìíỉĩịòóỏõọôốồổỗộơớờởỡợùúủũụưứừửữựỳýỷỹỵđĐ]/g, '');

                if (
                    text.length >= 1 &&
                    text.length <= 6 &&
                    !['ĐIỂM', 'Bỏ qua', 'Rời phòng', 'Copy Link', 'Văn phòng làm việc', 'Gom Từ Online', 's', 'ế', 'ỗ', 'Ế', 'Ỗ'].includes(text)
                ) {
                    const rect = el.getBoundingClientRect();
                    const style = window.getComputedStyle(el);

                    const isVisible =
                        style.display !== 'none' &&
                        style.visibility !== 'hidden' &&
                        parseFloat(style.opacity) > 0 &&
                        (typeof el.checkVisibility !== 'function' || el.checkVisibility({ checkOpacity: true, checkVisibilityCSS: true }));

                    if (
                        isVisible &&
                        rect.width > 0 &&
                        rect.height > 0 &&
                        rect.top > 100 &&
                        rect.top < window.innerHeight - 150 &&
                        (style.borderRadius !== '0px' || parseFloat(style.fontSize) >= 14)
                    ) {
                        if (!detectedWords.includes(text)) detectedWords.push(text);
                    }
                }
            }
        }

        detectedWords = detectedWords.slice(0, 16);
        const currentKey = detectedWords.join(',');
        if (detectedWords.length === 0 || currentKey === gtLastKey) return;
        gtLastKey = currentKey;

        const gridEl = document.getElementById('wd-grid-syllables');
        if (gridEl) {
            const slots = detectedWords.slice(0, 16);
            while (slots.length < 16) slots.push(null);
            gridEl.innerHTML = slots.map((w, i) => w
                ? `<div class="wd-syllable" data-idx="${i}" data-word="${w.toLowerCase()}">${w}</div>`
                : `<div class="wd-syllable wd-empty-tile" data-idx="${i}" data-word="">?</div>`
            ).join('');
        }

        const comboEl = document.getElementById('wd-combo-words');
        if (comboEl && dictReady) {
            const combos = gtFindValidCombos(detectedWords);
            comboEl.innerHTML = combos.length
                ? combos.map(c => `<div class="wd-combo" data-combo="${c.toLowerCase()}">${c}</div>`).join('')
                : '<span class="wd-empty-msg">Không tìm thấy từ ghép hợp lệ</span>';

            comboEl.querySelectorAll('.wd-combo').forEach(chip => {
                chip.addEventListener('click', () => {
                    const parts = chip.dataset.combo.split(' ');
                    const tiles = gridEl.querySelectorAll('.wd-syllable');
                    tiles.forEach(t => t.classList.remove('wd-lit'));
                    const used = new Set();
                    parts.forEach(p => {
                        for (const t of tiles) {
                            if (t.dataset.word === p && !used.has(t)) {
                                t.classList.add('wd-lit');
                                used.add(t);
                                break;
                            }
                        }
                    });
                });
            });
        }
    }

    function attachDrag(panel, header) {
        const saved = loadPos();
        if (saved && saved.left && saved.top) {
            panel.style.setProperty('left', saved.left, 'important');
            panel.style.setProperty('top', saved.top, 'important');
            panel.style.setProperty('right', 'auto', 'important');
        }

        let dragging = false, ox = 0, oy = 0;
        function start(x, y, target) {
            if (target.closest('#wd-toggle') || target.closest('.wd-btn')) return false;
            const rect = panel.getBoundingClientRect();
            ox = x - rect.left;
            oy = y - rect.top;
            panel.style.setProperty('left', rect.left + 'px', 'important');
            panel.style.setProperty('top', rect.top + 'px', 'important');
            panel.style.setProperty('right', 'auto', 'important');
            return true;
        }
        function move(x, y) {
            const w = panel.offsetWidth, h = panel.offsetHeight;
            let left = Math.max(4, Math.min(window.innerWidth - w - 4, x - ox));
            let top = Math.max(4, Math.min(window.innerHeight - h - 4, y - oy));
            panel.style.setProperty('left', left + 'px', 'important');
            panel.style.setProperty('top', top + 'px', 'important');
            savePos(panel.style.left, panel.style.top);
        }
        header.addEventListener('mousedown', e => { dragging = start(e.clientX, e.clientY, e.target); });
        document.addEventListener('mousemove', e => { if (dragging) move(e.clientX, e.clientY); });
        document.addEventListener('mouseup', () => dragging = false);
        header.addEventListener('touchstart', e => {
            dragging = start(e.touches[0].clientX, e.touches[0].clientY, e.target);
        }, { passive: true });
        panel.addEventListener('touchmove', e => {
            if (!dragging) return;
            move(e.touches[0].clientX, e.touches[0].clientY);
            e.preventDefault();
        }, { passive: false });
        panel.addEventListener('touchend', () => dragging = false);
    }

    // Tự sao chép từ cuối (âm tiết thứ 2) của từ gợi ý vào clipboard
    let lastCopied = '';
    let lastCopiedFull = '';
    function copyLastWord(full) {
        const last = (full || '').trim().split(/\s+/).pop();
        if (!last) return;
        lastCopied = last;
        lastCopiedFull = (full || '').trim();
        const fallback = () => {
            try {
                const active = document.activeElement;
                const ta = document.createElement('textarea');
                ta.value = last;
                ta.setAttribute('readonly', '');
                ta.style.cssText = 'position:fixed;top:0;left:0;opacity:0;pointer-events:none;';
                document.body.appendChild(ta);
                ta.select();
                document.execCommand('copy');
                ta.remove();
                if (active && active.focus) active.focus();
            } catch (e) {}
        };
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(last).catch(fallback);
        } else {
            fallback();
        }
    }

    // Chạm vào gợi ý: CHỈ copy từ cuối. Không ghi nhận vào chuỗi (chạm chỉ để xem/copy, chưa chắc đã đánh);
    // vị trí trong chuỗi chỉ được cập nhật khi từ thật sự được gửi (hook Enter/nút gửi) hoặc hiện trên màn hình game.
    function chooseWord(full) {

    // Quy giá trị trong ô nhập về từ đầy đủ (nếu chỉ dán âm tiết cuối vừa copy thì lấy lại từ đầy đủ)
    function resolveInputWord(value) {
        const norm = normWord(value);
        if (!norm) return null;
        const n = norm.split(' ').length;
        if (n === 2) return norm;
        if (n === 1 && lastCopiedFull && normWord(lastCopiedFull).split(' ')[1] === norm) {
            return normWord(lastCopiedFull);
        }
        return null;
    }

    function fillAndSubmit(full) {
        lastSubmitted = full;

        noteChainWord(full);
        copyLastWord(full);

        if (!input) return;
        const setter = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value')?.set;
        if (setter) setter.call(input, full);
        else input.value = full;
        input.dispatchEvent(new Event('input', { bubbles: true }));
        input.focus();

        setTimeout(() => {
            let sendBtn =
                document.querySelector('button[class*="submitBtn"], button[class*="submit"]') ||
                Array.from(document.querySelectorAll('button')).find(b =>
                    b.querySelector('svg') ||
                    /gửi|submit|ok|xác nhận|gửi từ/i.test((b.textContent || '').trim())
                );
            if (sendBtn) {
                sendBtn.click();
            } else {
                ['keydown', 'keypress', 'keyup'].forEach(type => {
                    input.dispatchEvent(new KeyboardEvent(type, {
                        key: 'Enter', code: 'Enter', keyCode: 13, which: 13, bubbles: true, cancelable: true
                    }));
                });
            }
        }, 60);
    }

    function tryAutoPlay() {
        if (!autoEnabled) return;
        clearTimeout(autoTimer);

    function chip(full, scoreLabel, isBest) {
        const second = full.split(' ')[1];
        const badge = (scoreLabel !== undefined && scoreLabel !== null)

    // ===== PHÂN TÍCH CHUỖI DÀI NHẤT (bỏ hẳn từ cụt) =====
    // Mục tiêu: chọn nước đi mở ra chuỗi nối được NHIỀU từ nhất.
    // Với mỗi ứng viên (đã loại từ cụt) chạy DFS tìm đường đi dài nhất trong đồ thị từ
    // (ưu tiên nhánh ít đường ra trước — quy tắc Warnsdorff — để không tự bịt đường).
    const CHAIN_NODE_LIMIT = 1500;  // số nút tối đa cho mỗi ứng viên
    const CHAIN_TIME_LIMIT = 200;   // ms tối đa cho mỗi lần phân tích
    const CHAIN_MAX_CAND = 30;      // số ứng viên tối đa được phân tích
    const CHAIN_BRANCH = 4;         // số nhánh tối đa xét ở mỗi nút

    function nextOptions(word, usedSet) {
        const list = byFirst.get(word.split(' ')[1]);
        if (!list) return [];
        return list.filter(w => w !== word && !badWords.has(w) && !usedSet.has(w));
    }

    function staticDegree(word) {
        const list = byFirst.get(word.split(' ')[1]);
        return list ? list.length : 0;
    }

    // Độ dài chuỗi dài nhất tìm được, bắt đầu từ "start" (tính cả start)
    function longestChain(start, usedSet, nodeLimit) {
        const used = new Set(usedSet);
        used.add(start);
        let best = 1, nodes = 0;

        function dfs(word, len) {
            if (len > best) best = len;
            if (nodes++ >= nodeLimit) return;
            const opts = nextOptions(word, used);
            if (!opts.length) return;

            const ranked = opts
                .map(w => { const d = staticDegree(w); return { w, k: d === 0 ? Infinity : d }; })
                .sort((a, b) => a.k - b.k)
                .slice(0, CHAIN_BRANCH);

            for (const r of ranked) {
                used.add(r.w);
                dfs(r.w, len + 1);
                used.delete(r.w);
                if (nodes >= nodeLimit) return;
            }
        }

        dfs(start, 1);
        return best;
    }

    // Nhận danh sách nước đi ĐÃ loại từ cụt, trả về nước tốt nhất + bảng xếp hạng
    function analyzeChains(candidates) {
        const t0 = performance.now();
        const quick = candidates
            .map(w => ({ w, deg: nextOptions(w, usedWords).length }))
            .sort((a, b) => b.deg - a.deg)
            .slice(0, CHAIN_MAX_CAND);

        const scored = [];
        for (const c of quick) {
            if (scored.length >= 6 && performance.now() - t0 > CHAIN_TIME_LIMIT) break;
            scored.push({ w: c.w, deg: c.deg, score: longestChain(c.w, usedWords, CHAIN_NODE_LIMIT) });
        }
        scored.sort((a, b) => (b.score - a.score) || (b.deg - a.deg));

        const top = scored[0];
        return {
            word: top.w,
            kind: 'ok',
            score: top.score,
            reason: `⛓ Chuỗi ước tính ≥ ${top.score} từ · còn ${top.deg} đường nối · không cụt`,
            alternatives: scored.slice(1, 4).map(s => ({ w: s.w, score: s.score })),
            all: scored
        };
    }

    // Gọi mỗi khi thấy/đánh một từ: chạm "học hành" => bắt chuỗi; sau đó theo dõi vị trí trong chuỗi
    // Theo dõi TUẦN TỰ: chỉ tiến đúng 1 bước khi gặp từ kế tiếp (số n -> n+1).
    // Từ chuỗi lệch thứ tự (nhảy xa / lùi) KHÔNG làm đổi số ngay; chỉ đồng bộ lại khi gặp tiếp
    // từ liền sau nó trong chuỗi (xác nhận 2 từ liên tiếp) => số không bị nhảy lung tung.
    function noteChainWord(w) {
        const k = toneKey(w);
        if (k === CHAIN_HEAD && !chainState.caught) {
            chainState.caught = true;
            chainState.idx = 0;
            chainState.pending = -1;
        } else if (chainState.caught && chainIdx.has(k)) {
            const pos = chainIdx.get(k)[0];
            if (pos === chainState.idx) {
                // cùng từ được thấy lại (màn hình + gửi) => giữ nguyên số
            } else if (pos === chainState.idx + 1) {
                chainState.idx = pos;
                chainState.pending = -1;
            } else if (chainState.pending >= 0 && pos === chainState.pending + 1) {
                chainState.idx = pos;               // 2 từ liên tiếp trong chuỗi => đồng bộ lại
                chainState.pending = -1;
            } else {
                chainState.pending = pos;           // chờ xác nhận, chưa đổi số
            }
        }
        if (chainState.caught) showCatchBanner();
    }

    // Khoảng cách (số từ) từ mỗi âm tiết tới âm tiết "học", chỉ đi qua từ KHÔNG thuộc chuỗi
    function distToHead(usedSet) {
        const dist = new Map([[CHAIN_HEAD_SYL, 0]]);
        const q = [CHAIN_HEAD_SYL];
        for (let h = 0; h < q.length; h++) {
            const sy = q[h], d = dist.get(sy);
            for (const e of (byLast.get(sy) || [])) {
                if (dist.has(e.p)) continue;
                if (chainDictWords.has(e.w) || badWords.has(e.w) || usedSet.has(e.w)) continue;
                dist.set(e.p, d + 1);
                q.push(e.p);
            }
        }
        return dist;
    }

    function leadMove(base, word, reason, tag) {
        const rest = base ? base.all.filter(x => x.w !== word) : [];
        return {
            word, kind: 'ok', score: tag, reason,
            alternatives: rest.slice(0, 3).map(x => ({ w: x.w, score: x.score })),
            all: [{ w: word, score: tag }, ...rest]
        };
    }

    // Chọn nước đi theo 2 giai đoạn: (1) tránh chuỗi + dẫn tới "học hành"; (2) đã bắt được thì đi theo chuỗi
    function pickMove(available, safe) {
        if (!chainState.caught) {
            const headWord = keyToWord.get(CHAIN_HEAD);
            const pool = safe.filter(w => !chainDictWords.has(w));
            if (headWord && available.includes(headWord)) {
                return { best: leadMove(pool.length ? analyzeChains(pool) : null, headWord, '🎯 Nối vào "' + headWord + '" — bắt chuỗi!', '🎯') };
            }
            if (!pool.length) return { best: null, msg: 'Chỉ còn từ thuộc chuỗi — chưa được dùng (chờ tới "học hành")' };
            const dist = distToHead(usedWords);
            let minD = Infinity;
            const scored = pool.map(w => {
                const d = dist.get(toneSyl(w.split(' ')[1]));
                const dd = d === undefined ? Infinity : d;
                if (dd < minD) minD = dd;
                return { w, d: dd };
            });
            if (minD === Infinity) {
                const b = analyzeChains(pool);
                b.reason = '⚠ Chưa thấy đường tới "học" · ' + b.reason;
                return { best: b };
            }
            const b = analyzeChains(scored.filter(x => x.d === minD).map(x => x.w));
            b.reason = (minD === 0 ? '🎯 Kết bằng "học" — chờ nối "học hành"' : '🎯 Dẫn tới "học hành": còn ' + minD + ' bước') + ' · tránh từ trong chuỗi';
            return { best: b };
        }
        const opts = safe.filter(w => chainDictWords.has(w))
            .map(w => ({ w, i: chainIdx.get(toneKey(w))[0] }))
            .filter(o => o.i > chainState.idx)          // chỉ đi tiến, không lùi số
            .sort((a, b) => a.i - b.i);                 // gần nhất (n+1) trước
        const base = analyzeChains(safe);
        if (opts.length) {
            const nxt = opts[0];
            return { best: leadMove(base, nxt.w, '⛓ ' + (nxt.i + 1) + '. ' + CHAIN_WORDS[nxt.i] + ' (' + (nxt.i + 1) + '/' + CHAIN_LEN + ')', '⛓') };
        }
        base.reason = '⛓ Hết đường theo chuỗi · ' + base.reason;
        return { best: base };
    }

    function renderSuggestions(text) {
        const box = document.getElementById('wd-suggest');
        if (!box) return;

        if (!dictReady) {
            box.innerHTML = `<div class="wd-empty-msg" style="color:#fbbf24;">Đang tải kho từ...</div>`;
            updateAISuggestion(null);
            updateMiniWord();
            return;
        }

        const norm = normWord(text);
        const parts = norm.split(' ');
        if (parts.length !== 2) {
            box.innerHTML = `<div class="wd-empty-msg">Từ không hợp lệ (2 âm tiết)</div>`;
            updateAISuggestion(null);
            updateMiniWord();
            return;
        }

        const key = parts[1];
        const available = (byFirst.get(key) || []).filter(w => w !== norm && !badWords.has(w) && !usedWords.has(w));

        if (!available.length) {
            box.innerHTML = `<div class="wd-empty-msg" style="color:#f87171;">Hết từ nối cho "${key}"</div>`;
            updateAISuggestion(null);
            updateMiniWord();
            if (lastTrapAlertKey !== norm) {
                lastTrapAlertKey = norm;
            }
            if (autoEnabled && lastSkippedWord !== norm) {
                lastSkippedWord = norm;
                clearTimeout(skipTimer);
                skipTimer = setTimeout(() => {
                    if (autoEnabled && lastText && normWord(lastText) === norm) {
                        sendSkip();
                    }
                }, 300 + Math.random() * 300);
            }
            return;
        }

        // ẨN từ cụt: không hiển thị, không gợi ý, không tự động đánh
        const safe = available.filter(w => !isDeadEndWord(w, badWords));
        const hidden = available.length - safe.length;

        if (!safe.length) {
            box.innerHTML = `<div class="wd-empty-msg" style="color:#fbbf24;">Chỉ còn từ cụt (đã ẩn ${hidden}) — không có nước nối tiếp</div>`;
            updateAISuggestion(null, 'Chỉ còn từ cụt');
            return;
        }

        const pm = pickMove(available, safe);
        if (!pm.best) {
            box.innerHTML = `<div class="wd-empty-msg" style="color:#fbbf24;">${pm.msg}</div>`;
            updateAISuggestion(null, pm.msg);
            return;
        }
        const best = pm.best;
        updateAISuggestion(best);

        // Chip xếp theo độ dài chuỗi giảm dần; chạm = gửi từ (không còn xem trước từ kế tiếp)
        box.innerHTML = best.all.slice(0, 16).map((s, i) => chip(s.w, s.score, i === 0)).join('') +
            (hidden ? `<div class="wd-empty-msg" style="width:100%;text-align:center;font-size:10px;">Đã ẩn ${hidden} từ cụt</div>` : '');

        box.querySelectorAll('.wd-sugg').forEach(el => {
            el.onclick = () => {
                chooseWord(el.dataset.full);
                el.style.background = 'linear-gradient(135deg,#10b981,#059669)';
                el.style.color = '#fff';
            };
        });

        if (autoEnabled) tryAutoPlay();
    }

    function scanWord() {
        initUI();
        if (chainState.caught && !document.getElementById('wd-catch-banner')) showCatchBanner();

        const allTextElements = document.querySelectorAll('*');
        let matchedText = "";

        for (let el of allTextElements) {
            if (el.closest('#wd-mobile-panel')) continue;
            if (el.children.length === 0 && el.innerText) {
                const text = el.innerText.trim();

                if (
                    text.length >= 2 &&
                    !text.includes('\n') &&
                    !['ĐIỂM', 'Đối thủ', 'Bỏ qua', 'ĐÃ NỐI', 'esc', 'Bot', 'Chơi', 'nhóm'].some(k => text.includes(k)) &&
                    !/^\d+$/.test(text)
                ) {
                    const style = window.getComputedStyle(el);
                    if (parseFloat(style.fontSize) >= 20 && el.getBoundingClientRect().height > 0) {
                        matchedText = text;
                    }
                }
            }
        }

        if (matchedText && matchedText !== lastText) {
            lastText = matchedText;

            const norm = normWord(matchedText);
            if (norm && norm.split(' ').length === 2) {
                usedWords.add(norm);
                noteChainWord(norm);
            }

            const origEl = document.getElementById('wd-original');
            if (origEl) {
                origEl.innerText = matchedText;
            }

            renderSuggestions(matchedText);
        }
    }

    // Công cụ Console:
    //   wdBridge('lượng','vi')  -> tìm đường ngắn nhất "lượng ... vi" qua kho từ (bỏ qua từ đã thuộc chuỗi)
    //   wdCheckChain()          -> liệt kê từ trong chuỗi mà kho từ KHÔNG có (để sửa)
    window.wdBridge = function (a, b, maxLen) {
        a = normWord(a); b = normWord(b); maxLen = maxLen || 8;
        if (!dictReady) { console.log('Kho từ chưa tải xong, thử lại sau vài giây.'); return null; }
        const nexts = x => [...new Set((byFirst.get(x) || []).filter(w => !chainDictWords.has(w)).map(w => w.split(' ')[1]))];
        const dist = new Map([[a, 0]]), parents = new Map();
        let frontier = [a];
        for (let d = 1; d <= maxLen && frontier.length && !dist.has(b); d++) {
            const nf = [];
            for (const x of frontier) for (const y of nexts(x)) {
                if (!dist.has(y)) { dist.set(y, d); parents.set(y, [x]); nf.push(y); }
                else if (dist.get(y) === d) parents.get(y).push(x);
            }
            frontier = nf;
        }
        if (!dist.has(b)) { console.log('Không tìm thấy đường từ "' + a + '" tới "' + b + '" trong ' + maxLen + ' bước.'); return []; }
        const out = [];
        (function walk(y, path) {
            if (out.length >= 15) return;
            if (y === a) { out.push([a, ...path].join(' ')); return; }
            for (const x of parents.get(y)) walk(x, [y, ...path]);
        })(b, []);
        // walk dựng ngược nên path đã đúng thứ tự: a ... b
        console.log('Đường ngắn nhất (' + dist.get(b) + ' từ):\n' + out.join('\n'));
        return out;
    };
    window.wdCheckChain = function () {
        if (!dictReady) { console.log('Kho từ chưa tải xong.'); return null; }
        const missing = [];
        for (let i = 1; i < CHAIN_SYL.length; i++) {
            const w = normWord(CHAIN_SYL[i - 1] + ' ' + CHAIN_SYL[i]);
            if (!keyToWord.has(toneKey(w))) missing.push((i - 1) + ': ' + w);
        }
        console.log(missing.length ? 'Từ trong chuỗi KHÔNG có trong kho:\n' + missing.join('\n') : 'Mọi từ trong chuỗi đều có trong kho.');
        return missing;
    };
